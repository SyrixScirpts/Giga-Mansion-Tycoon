-- Draggable Logic
local dragging = false
local dragInput, dragStart, startPos

HeaderFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

HeaderFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Helper to create option rows
local function createToggleRow(name, positionY, initialValue, callback)
    local RowFrame = Instance.new("Frame")
    RowFrame.Name = name .. "Row"
    RowFrame.Size = UDim2.new(1, -24, 0, 36)
    RowFrame.Position = UDim2.new(0, 12, 0, positionY)
    RowFrame.BackgroundTransparency = 1
    RowFrame.Parent = ContentFrame

    local Label = Instance.new("TextLabel")
    Label.Name = "Label"
    Label.Size = UDim2.new(1, -40, 1, 0)
    Label.BackgroundTransparency = 1
    Label.Text = name
    Label.TextColor3 = Color3.fromRGB(255, 255, 255)
    Label.TextSize = 18
    Label.Font = Enum.Font.FredokaOne
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = RowFrame

    local CheckboxBtn = Instance.new("TextButton")
    CheckboxBtn.Name = "Checkbox"
    CheckboxBtn.Size = UDim2.new(0, 26, 0, 26)
    CheckboxBtn.Position = UDim2.new(1, -26, 0.5, -13)
    CheckboxBtn.BackgroundColor3 = initialValue and Color3.fromRGB(240, 50, 50) or Color3.fromRGB(20, 20, 20)
    CheckboxBtn.BorderSizePixel = 0
    CheckboxBtn.Text = initialValue and "✓" or ""
    CheckboxBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    CheckboxBtn.TextSize = 17
    CheckboxBtn.Font = Enum.Font.GothamBold
    CheckboxBtn.Parent = RowFrame

    local BoxCorner = Instance.new("UICorner")
    BoxCorner.CornerRadius = UDim.new(0, 4)
    BoxCorner.Parent = CheckboxBtn

    local BoxStroke = Instance.new("UIStroke")
    BoxStroke.Color = Color3.fromRGB(65, 65, 65)
    BoxStroke.Thickness = 1.5
    BoxStroke.Parent = CheckboxBtn

    local state = initialValue

    local function updateState(newState)
        state = newState
        if state then
            CheckboxBtn.BackgroundColor3 = Color3.fromRGB(240, 50, 50)
            CheckboxBtn.Text = "✓"
        else
            CheckboxBtn.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
            CheckboxBtn.Text = ""
        end
        pcall(callback, state)
    end

    CheckboxBtn.MouseButton1Click:Connect(function()
        updateState(not state)
    end)
    
    Label.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            updateState(not state)
        end
    end)

    return updateState
end

-- Tycoon Helper Functions
local function getMyTycoon()
    local tycoons = workspace:FindFirstChild("Tycoons")
    if not tycoons then return nil end
    for _, t in ipairs(tycoons:GetChildren()) do
        local owner = t:GetAttribute("Owner")
        if owner == LocalPlayer.Name or string.find(t.Name, LocalPlayer.Name) then
            return t
        end
    end
    return nil
end

local function getCheapestAffordableButton(tycoon, money)
    if not tycoon then return nil end
    local buttons = {}
    
    local function scan(folder)
        for _, item in ipairs(folder:GetChildren()) do
            if item:IsA("Model") then
                local price = item:GetAttribute("Price") or (item:FindFirstChild("Price") and item.Price.Value)
                if price then
                    table.insert(buttons, {
                        model = item,
                        price = tonumber(price) or 0
                    })
                end
            elseif item:IsA("Folder") then
                scan(item)
            end
        end
    end
    
    if tycoon:FindFirstChild("Buttons") then scan(tycoon.Buttons) end
    if tycoon:FindFirstChild("Extensions") then scan(tycoon.Extensions) end
    
    table.sort(buttons, function(a, b) return a.price < b.price end)
    
    for _, btnData in ipairs(buttons) do
        if btnData.price <= money then
            return btnData
        end
    end
    return nil
end

-- Auto Collect Cash Loop (No Teleport)
task.spawn(function()
    while _G.GigaTycoonScriptLoaded do
        task.wait(0.8)
        if AutoCollectEnabled then
            pcall(function()
                local tycoon = getMyTycoon()
                if tycoon then
                    local bank = tycoon.Models:FindFirstChild("Bank")
                    local zonePart = bank and bank:FindFirstChild("ZonePart")
                    local char = LocalPlayer.Character
                    local hrp = char and char:FindFirstChild("HumanoidRootPart")
                    if zonePart and hrp and typeof(firetouchinterest) == "function" then
                        firetouchinterest(hrp, zonePart, 0)
                        task.wait(0.05)
                        firetouchinterest(hrp, zonePart, 1)
                    end
                end
            end)
        end
    end
end)

-- Auto Buy Least Cash Buttons Loop
task.spawn(function()
    while _G.GigaTycoonScriptLoaded do
        task.wait(0.4)
        if AutoBuyEnabled then
            pcall(function()
                local tycoon = getMyTycoon()
                local char = LocalPlayer.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
                local money = leaderstats and leaderstats:FindFirstChild("Money") and leaderstats.Money.Value or 0
                
                if tycoon and hrp and typeof(firetouchinterest) == "function" then
                    local btnData = getCheapestAffordableButton(tycoon, money)
                    if btnData and btnData.model then
                        local model = btnData.model
                        local press = model:FindFirstChild("Press") or model:FindFirstChild("Plate") or model:FindFirstChildWhichIsA("BasePart")
                        if press then
                            firetouchinterest(hrp, press, 0)
                            task.wait(0.05)
                            firetouchinterest(hrp, press, 1)
                        end
                    end
                end
            end)
        end
    end
end)

-- Anti AFK Loop (Jump every 18 mins if standing still; resets timer on walk)
task.spawn(function()
    local idleSeconds = 0
    local lastPosition = nil

    while _G.GigaTycoonScriptLoaded do
        task.wait(1.0)
        if AntiAFKEnabled then
            pcall(function()
                local char = LocalPlayer.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                local hrp = char and char:FindFirstChild("HumanoidRootPart")

                if hum and hrp then
                    local currentPos = hrp.Position
                    local isMoving = (hum.MoveDirection.Magnitude > 0.05) or (lastPosition and (currentPos - lastPosition).Magnitude > 0.5)
                    lastPosition = currentPos

                    if isMoving then
                        -- Reset timer whenever player walks
                        idleSeconds = 0
                    else
                        -- Player is standing still
                        idleSeconds = idleSeconds + 1
                        
                        -- 18 minutes = 1080 seconds
                        if idleSeconds >= 1080 then
                            -- Trigger Jump & Virtual Activity
                            hum.Jump = true
                            hum:ChangeState(Enum.HumanoidStateType.Jumping)
                            pcall(function()
                                VirtualUser:CaptureController()
                                VirtualUser:ClickButton2(Vector2.new())
                            end)
                            
                            -- Reset timer back to 0
                            idleSeconds = 0
                        end
                    end
                else
                    idleSeconds = 0
                end
            end)
        else
            idleSeconds = 0
        end
    end
end)
